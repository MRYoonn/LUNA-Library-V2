cmake_minimum_required(VERSION 3.21)
project(LunaAccuracySnapshotNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core)
enable_testing()
add_executable(luna_accuracy_snapshot_native main.cpp)
target_link_libraries(luna_accuracy_snapshot_native PRIVATE Qt6::Core)
add_test(NAME luna_accuracy_snapshot_native COMMAND luna_accuracy_snapshot_native)
set_tests_properties(luna_accuracy_snapshot_native PROPERTIES TIMEOUT 30)
