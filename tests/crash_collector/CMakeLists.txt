cmake_minimum_required(VERSION 3.24)
project(LunaCrashCollectorFixture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT WIN32)
  message(FATAL_ERROR "Actual Windows required")
endif()
add_executable(LUNA WIN32 main.cpp)
target_link_libraries(LUNA PRIVATE user32 shell32)
target_compile_definitions(LUNA PRIVATE UNICODE _UNICODE)
if(MSVC)
  target_compile_options(LUNA PRIVATE /Zi /Od /utf-8)
  target_link_options(LUNA PRIVATE /DEBUG:FULL)
endif()
