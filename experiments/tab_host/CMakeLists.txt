cmake_minimum_required(VERSION 3.24)
project(LunaTabHostProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
if(NOT WIN32)
    message(FATAL_ERROR "This experiment tests Windows foreign-window embedding only")
endif()
add_executable(luna-tab-host main.cpp)
target_link_libraries(luna-tab-host PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets user32)
if(MSVC)
    target_compile_options(luna-tab-host PRIVATE /utf-8)
endif()
