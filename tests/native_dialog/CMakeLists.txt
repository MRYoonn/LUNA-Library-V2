cmake_minimum_required(VERSION 3.24)
project(LunaNativeDialogProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network)
add_executable(luna-native-dialog main.cpp)
target_include_directories(luna-native-dialog PRIVATE ../../src)
target_link_libraries(luna-native-dialog PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
if(MSVC)
    target_compile_options(luna-native-dialog PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME native_save_dialog COMMAND luna-native-dialog)
set_tests_properties(native_save_dialog PROPERTIES TIMEOUT 20)
add_executable(luna-native-thumbnails thumbnails.cpp)
target_include_directories(luna-native-thumbnails PRIVATE ../../src)
target_link_libraries(luna-native-thumbnails PRIVATE Qt6::Core Qt6::Gui)
if(MSVC)
    target_compile_options(luna-native-thumbnails PRIVATE /utf-8)
endif()
add_test(NAME native_thumbnail_cache COMMAND luna-native-thumbnails)
set_tests_properties(native_thumbnail_cache PROPERTIES TIMEOUT 30)
add_executable(luna-native-filemoves filemoves.cpp)
target_include_directories(luna-native-filemoves PRIVATE ../../src)
target_link_libraries(luna-native-filemoves PRIVATE Qt6::Core)
if(MSVC)
    target_compile_options(luna-native-filemoves PRIVATE /utf-8)
endif()
add_test(NAME native_file_moves COMMAND luna-native-filemoves)
set_tests_properties(native_file_moves PROPERTIES TIMEOUT 20)
add_executable(luna-native-documentclaims documentclaims.cpp)
target_include_directories(luna-native-documentclaims PRIVATE ../../src)
target_link_libraries(luna-native-documentclaims PRIVATE Qt6::Core Qt6::Network)
if(MSVC)
    target_compile_options(luna-native-documentclaims PRIVATE /utf-8)
endif()
add_test(NAME native_document_claims COMMAND luna-native-documentclaims)
set_tests_properties(native_document_claims PROPERTIES TIMEOUT 20)
