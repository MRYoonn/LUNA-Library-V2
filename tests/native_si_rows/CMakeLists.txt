cmake_minimum_required(VERSION 3.21)
project(LunaSiRowsNativeTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core)
enable_testing()
add_executable(luna_si_rows_native main.cpp)
target_include_directories(luna_si_rows_native PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src")
target_link_libraries(luna_si_rows_native PRIVATE Qt6::Core)
add_test(NAME luna_si_rows_native COMMAND luna_si_rows_native)
set_tests_properties(luna_si_rows_native PROPERTIES TIMEOUT 30)
